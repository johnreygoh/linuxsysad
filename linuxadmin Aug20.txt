// swap space
exceed ram usage -> space in storage

recommend:
-do not modify existing swap space
-select another partition as swap space

// check swap space
swapon --show
swapon -s
free -h

// if changing swap partitions, turn off
sudo swapoff -a

// create swapfile
fallocate -l 2G /mnt/partition1/swapfile

//set permissions to swapfile
sudo chmod 600 /mnt/partition1/swapfile

//make it swap and enable
sudo mkswap /mnt/partition1/swapfile
sudo swapon /mnt/partition1/swapfile

//to make it permanent, add to /etc/fstab
/mnt/partition1/swapfile none swap sw 0 0 

//remove swapfile from swap space
sudo swapoff -a
rm /mnt/partition1/swapfile
nano /etc/fstab		(remove swap entry)

cron and at

// task
sudo nano script2.sh

#!/bin/bash
echo "script2 executed on: $(date)" >> /home/john/script2.log


//cron to sched
//existing jobs
crontab -l

//edit crontab (to add a job)
crontab -e

//crontab structure
//syntax
min hour dayofmonth month dayofweek command

//examples
//run script every 5 minutes
*/5 * * * * /home/john/script2.sh

//run everyday at 2am
0 2 * * * /home/john/script2.sh

//run every Sunday at midnight
0 0 * * 0 /home/john/script2.sh

//7:30am Jan 1 Monday
30 7 1 1 1 script2.sh

//shortcuts for cron (use inside crontab)
@reboot		run once at startup
@hourly		0 * * * * /script
@daily		0 0 * * * /script
@monthly	0 0 1 * * /script
@weekly		0 0 * * 0 /script

//terminal
@monthly /home/john/script2.sh

//crond service
systemctl status crond
systemctl start crond
systemctl enable crond
systemctl restart crond

// check cron jobs executed (last hour)
journalctl -u crond --since "1 hour ago"

//AT command (1-time)

script2.log | at 1:00 AM December 25

// sharing of folder made in a file/config
/etc/samba/smb.conf

// backup orig config file
sudo cp /etc/samba/smb.conf /etc/samba/smb.conf.orig

// create shared folder(s)
sudo mkdir -p /srv/samba/everyone/johnfolder
sudo chmod 777 /srv/samba/everyone/johnfolder
sudo chown -R nobody:nobody /srv/samba/everyone/johnfolder

// add the shared folder to smb.conf
[johnfolder]
        comment = Shared folder
        path = /srv/samba/everyone/johnfolder
        browseable = yes
        writable = yes
        guest ok = yes
        read only = no
        force user = nobody
